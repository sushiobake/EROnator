  const renderTagGrid = (placement: 'inBoard' | 'below') => {
    const TAG_SELECT_BLUE = '#3b82f6';
    const isBelow = placement === 'below';
    const TagButton = ({
      tagKey,
      displayName,
      checked,
      onToggle,
    }: {
      tagKey: string;
      displayName: string;
      checked: boolean;
      onToggle: () => void;
    }) => (
      <button
        type="button"
        onClick={onToggle}
        style={{
          ...(isMobile
            ? mobileQuizLikeChoiceStyle({
                display: 'flex',
                alignItems: 'center',
                justifyContent: 'flex-start',
                textAlign: 'left',
                backgroundColor: checked ? '#dbeafe' : '#faf8f5',
                color: checked ? '#1d4ed8' : 'var(--color-text)',
                border: `2px solid ${checked ? TAG_SELECT_BLUE : '#e5e7eb'}`,
                borderRadius: 8,
                cursor: 'pointer',
              })
            : {
                display: 'flex',
                alignItems: 'center',
                padding: '10px 12px',
                fontSize: 14,
                fontWeight: 500,
                textAlign: 'left',
                backgroundColor: checked ? '#dbeafe' : '#faf8f5',
                color: checked ? '#1d4ed8' : 'var(--color-text)',
                border: `2px solid ${checked ? TAG_SELECT_BLUE : '#e5e7eb'}`,
                borderRadius: 8,
                cursor: 'pointer',
              }),
        }}
      >
        <span style={{ flex: 1, overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{displayName}</span>
      </button>
    );

    const gridStyle: CSSProperties = {
      display: 'grid',
      gridTemplateColumns: isBelow ? 'repeat(2, minmax(0, 1fr))' : 'repeat(4, 1fr)',
      gap: isBelow ? 6 : 8,
      marginBottom: isBelow ? 0 : 14,
    };

    if (isFamousStep) {
      const opts = currentFamousOptions;
      const cat = (step as string).startsWith('f1') ? priorityOrder[0]! : (step as string).startsWith('f2') ? priorityOrder[1]! : priorityOrder[2]!;
      return opts.length === 0 ? (
        <p style={{ fontSize: 14, color: 'var(--color-text-muted)', marginBottom: 14 }}>このカテゴリにはタグがありません。そのまま次へ進んでね。</p>
      ) : (
        <div style={gridStyle}>
          {opts.map(opt => (
            <TagButton
              key={opt.tagKey}
              tagKey={opt.tagKey}
              displayName={opt.displayName}
              checked={checkedFamous.has(opt.tagKey)}
              onToggle={() => toggleFamous(opt.tagKey, opt.displayName, cat)}
            />
          ))}
        </div>
      );
    }
    const batch = currentUnknownBatch;
    return batch.length === 0 ? (
      <p style={{ fontSize: 14, color: 'var(--color-text-muted)', marginBottom: 14 }}>この中にはありません。そのまま次へ進んでね。</p>
    ) : (
      <div style={gridStyle}>
        {batch.map(t => (
          <TagButton
            key={t.tagKey}
            tagKey={t.tagKey}
            displayName={t.displayName}
            checked={checkedUnknown.has(t.tagKey)}
            onToggle={() => toggleUnknown(t.tagKey, t.displayName)}
          />
        ))}
      </div>
    );
  };

  const useMobileTagListBelow = isMobile && (isFamousStep || isBackStep);
  const mobileTagListBelow = useMobileTagListBelow ? (
    <div style={{ width: '100%', maxWidth: 480, margin: '0 auto', zoom: MOBILE_WB_ZOOM }}>{renderTagGrid('below')}</div>
  ) : undefined;

  return (
    <>
    <Stage
      characterVariant="usually"
      characterSpeech={
        <div style={isMobile ? { fontSize: 24, lineHeight: 1.3, textAlign: 'center' } : {}}>
          <p style={{ margin: 0, fontWeight: 600, color: 'var(--color-text)', fontSize: isMobile ? 24 : 17 }}>
            <span style={{ display: 'inline-flex', alignItems: 'center', justifyContent: 'center', width: isMobile ? 28 : 22, height: isMobile ? 28 : 22, backgroundColor: '#334155', color: '#fff', borderRadius: 6, fontSize: isMobile ? 14 : 12, fontWeight: 'bold', marginRight: 8, verticalAlign: 'middle' }}>
              {questionNumDisplay}
            </span>
            {instruction}
          </p>
        </div>
      }
      whiteboardWide={true}
      mobileBelowCanvas={mobileTagListBelow}
      mobileExtendWhiteboard={useMobileTagListBelow ? false : undefined}
    >
      <div style={{ padding: isMobile ? '0.5rem 0' : '0.55rem 0', maxWidth: '100%', minWidth: 0 }}>
        {!useMobileTagListBelow && renderTagGrid('inBoard')}
        <div
          style={{
            display: 'flex',
            flexDirection: isMobile ? 'column' : 'row',
            alignItems: isMobile ? 'stretch' : 'center',
            gap: isMobile ? 8 : 10,
            width: '100%',
            maxWidth: 640,
          }}
        >
          <div style={{ display: 'flex', gap: 10, flex: isMobile ? undefined : 1, minWidth: 0, alignItems: 'stretch', width: isMobile ? '100%' : undefined }}>
            <button
              type="button"
              aria-disabled={famousPrimaryDisabled}
              onClick={() => (isFamousStep ? goNextFromFamous() : goNextFromUnknown())}
              style={{
                flex: 1,
                minWidth: 0,
                ...(isMobile
                  ? mobileQuizLikeChoiceStyle({
                      display: 'flex',
                      alignItems: 'center',
                      justifyContent: 'center',
                      fontWeight: 600,
                    })
                  : { padding: '14px 18px', fontSize: 16, fontWeight: 600 }),
                backgroundColor: famousPrimaryDisabled ? '#e5e7eb' : '#3b82f6',
                color: famousPrimaryDisabled ? '#9ca3af' : '#fff',
                border: 'none',
                borderRadius: 12,
                cursor: famousPrimaryDisabled ? 'not-allowed' : 'pointer',
              }}
            >
              {rc.btnOk}
            </button>
            <button
              type="button"
              disabled={notInListDisabled}
              onClick={() => (isFamousStep ? goNextFromFamous(true) : goNextFromUnknown(true))}
              style={{
                flex: 1,
                minWidth: 0,
                ...(isMobile
                  ? mobileQuizLikeChoiceStyle({
                      display: 'flex',
                      alignItems: 'center',
                      justifyContent: 'center',
                      fontWeight: 600,
                      border: '2px solid #e5e7eb',
                      backgroundColor: notInListDisabled ? '#f3f4f6' : '#faf8f5',
                      color: notInListDisabled ? '#9ca3af' : 'var(--color-text)',
                    })
                  : {
                      padding: '14px 18px',
                      fontSize: 16,
                      fontWeight: 600,
                      border: '2px solid #e5e7eb',
                      backgroundColor: notInListDisabled ? '#f3f4f6' : '#faf8f5',
                      color: notInListDisabled ? '#9ca3af' : 'var(--color-text)',
                    }),
                borderRadius: 12,
                cursor: notInListDisabled ? 'not-allowed' : 'pointer',
              }}
            >
              {rc.btnNotInList}
            </button>
          </div>
          <button
            type="button"
            onClick={() => {
              const s = step as string;
              /** 今表示中の一覧のチェックだけ外す（確定済みタグは触らない） */
              if (['f1', 'f1_2', 'f2', 'f2_2', 'f3', 'f3_2'].includes(s)) {
                setFamousPickWarning(null);
                setCheckedFamous(new Set());
                return;
              }
              if (['s4', 's5', 's6', 's7', 's8'].includes(s)) {
                setCheckedUnknown(new Set());
              }
            }}
            style={{
              flexShrink: 0,
              alignSelf: isMobile ? 'flex-end' : 'center',
              padding: isMobile ? '5px 10px' : '6px 12px',
              fontSize: isMobile ? 11 : 12,
              fontWeight: 600,
              borderRadius: 8,
              cursor: 'pointer',
              boxSizing: 'border-box',
              border: '1px solid #d1d5db',
              backgroundColor: '#fff',
              color: 'var(--color-text-muted)',
              marginLeft: isMobile ? 0 : 4,
            }}
          >
            {rc.btnRetry}
          </button>
        </div>
        {(isFamousStep || isBackStep) && (
          <div style={{ marginTop: 6, width: '100%', display: 'flex', justifyContent: 'flex-end' }}>
            <button
              type="button"
              onClick={() => {
                const s = step as FrontStep | BackStep;
                setFamousPickWarning(null);
                if (s === 'f1') {
                  setFrontNavStack([]);
                  setCheckedFamous(new Set());
                  setStep('initial');
                } else if (isFamousStep) {
                  setFrontNavStack(prev => {
                    if (prev.length === 0) return prev;
                    const ent = prev[prev.length - 1]!;
                    setSelectedFamous(sel => {
                      const popped = sel.slice(-ent.undoCount);
                      setCheckedFamous(new Set(popped.map(x => x.tagKey)));
                      return sel.slice(0, -ent.undoCount);
                    });
                    setStep(ent.returnTo);
                    return prev.slice(0, -1);
                  });
                } else if (s === 's4') {
                  setStep('sort1');
                } else if (isBackStep) {
                  const prevMap: Record<string, BackStep> = { s5: 's4', s6: 's5', s7: 's6', s8: 's7' };
                  const prev = prevMap[s];
                  if (prev) {
                    const toRestore = selectedUnknown.slice(-lastAddedUnknownCount);
                    setSelectedUnknown(prevSel => prevSel.slice(0, -lastAddedUnknownCount));
                    setCheckedUnknown(new Set(toRestore.map(x => x.tagKey)));
                    setStep(prev);
                  }
                }
              }}
              style={{
                display: 'flex',
                alignItems: 'center',
                gap: 6,
                padding: '4px 10px',
                fontSize: 13,
                cursor: 'pointer',
                backgroundColor: 'transparent',
                border: 'none',
                borderRadius: 6,
                color: 'var(--color-text-muted)',
              }}
            >
              <svg style={{ width: 15, height: 15 }} viewBox="0 0 24 24" fill="currentColor">
                <path d="M7.83 11H20v2H7.83l5.59 5.59L12 20l-8-8 8-8 1.41 1.41L7.83 11z" />
              </svg>
              {rcFixBack}
            </button>
          </div>
        )}
        {isFamousStep && (
          <RecommendPickedTagsRow tags={frontPickedTagsDisplay} isMobile={isMobile} compact={useMobileTagListBelow} />
        )}
        {isBackStep && (
          <RecommendPickedTagsRow tags={backPickedTagsDisplay} isMobile={isMobile} compact={useMobileTagListBelow} />
        )}
